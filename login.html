import {createClient} from 'https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2/+esm'; import {SUPABASE_URL,SUPABASE_ANON_KEY} from './config.js'; const sb=createClient(SUPABASE_URL,SUPABASE_ANON_KEY); const form=document.getElementById('login-form'),message=document.getElementById('message'),submit=document.getElementById('submit'); document.getElementById('toggle').addEventListener('click',()=>{const input=document.getElementById('password');const show=input.type==='password';input.type=show?'text':'password';document.getElementById('toggle').textContent=show?'Ocultar':'Mostrar';}); form.addEventListener('submit',async e=>{e.preventDefault();submit.disabled=true;message.textContent='Verificando credenciais...';try{const {data,error}=await sb.auth.signInWithPassword({email:document.getElementById('email').value.trim(),password:document.getElementById('password').value});if(error)throw error;const {data:profile,error:profileError}=await sb.from('perfis').select('tipo,empresa_id').eq('id',data.user.id).single();if(profileError||!profile){message.textContent='Sua conta não possui um perfil autorizado. Contate a administração.';await sb.auth.signOut();return;}if(profile.tipo==='admin'&&profile.empresa_id===null){location.replace('./painel.html');return;}message.textContent='Sua conta foi autenticada, mas seu painel ainda está em desenvolvimento.';await sb.auth.signOut();}catch(err){console.error('Erro no login:',err);message.textContent='Não foi possível entrar. Verifique o e-mail e a senha e tente novamente.';}finally{submit.disabled=false;}});
